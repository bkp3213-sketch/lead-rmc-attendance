async function registerFace() { const employeeId = document.getElementById("employeeId").value.trim(); const employeeName = document.getElementById("employeeName").value.trim(); const status = document.getElementById("status"); const button = document.getElementById("captureBtn"); if (!employeeId || !employeeName) { status.innerHTML = "⚠️ Enter Employee ID and Employee Name."; return; } button.disabled = true; button.innerText = "SCANNING..."; status.innerHTML = "📷 Detecting face..."; try { const video = document.getElementById("video"); // Detect face const detection = await faceapi .detectSingleFace( video, new faceapi.TinyFaceDetectorOptions({ inputSize: 320, scoreThreshold: 0.5 }) ) .withFaceLandmarks(true) .withFaceDescriptor(); // No face if (!detection) { status.innerHTML = "❌ Face not detected.
" + "Keep your face clearly visible and try again."; button.disabled = false; button.innerText = "CAPTURE & REGISTER FACE"; return; } // Create face descriptor const descriptor = Array.from(detection.descriptor); status.innerHTML = "✅ Face detected.
" + "💾 Saving employee..."; // Fill hidden form document.getElementById( "formEmployeeId" ).value = employeeId; document.getElementById( "formEmployeeName" ).value = employeeName; document.getElementById( "formDescriptor" ).value = JSON.stringify(descriptor); // Get form ONCE const form = document.getElementById( "registrationForm" ); // Apps Script URL form.action = APPS_SCRIPT_URL; button.innerText = "SAVING..."; // Submit form.submit(); /* Apps Script saves the employee. The response is handled by the message event below. */ } catch (error) { console.error(error); status.innerHTML = "❌ Face processing error.
" + error.message; button.disabled = false; button.innerText = "CAPTURE & REGISTER FACE"; } }
